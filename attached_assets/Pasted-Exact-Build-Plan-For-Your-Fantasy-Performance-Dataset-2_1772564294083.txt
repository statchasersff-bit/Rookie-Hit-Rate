import nfl_data_py as nfl

years = list(range(2017, 2026))
seasonal = nfl.import_seasonal_data(years)

seasonal = seasonal[
    seasonal["position"].isin(["QB", "RB", "WR", "TE"])
]
3️⃣ Compute PPR + Half PPR Points
def compute_fantasy_points(df):
    df["fp_base"] = (
        df["passing_yards"] * 0.04 +
        df["passing_tds"] * 4 -
        df["interceptions"] * 2 +
        df["rushing_yards"] * 0.1 +
        df["rushing_tds"] * 6 +
        df["receiving_yards"] * 0.1 +
        df["receiving_tds"] * 6
    )

    df["fp_ppr"] = df["fp_base"] + df["receptions"]
    df["fp_half"] = df["fp_base"] + df["receptions"] * 0.5

    return df

seasonal = compute_fantasy_points(seasonal)
4️⃣ Remove Low-Usage Noise

Very important for clean ranks:

seasonal = seasonal[seasonal["games"] >= 8]

This prevents 1-game RB1 nonsense.

5️⃣ Compute Positional Ranks Per Season
def compute_pos_ranks(df, format_col):
    df = df.sort_values(
        ["season", "position", format_col],
        ascending=[True, True, False]
    )
    df[f"rank_{format_col}"] = (
        df.groupby(["season", "position"])
        .cumcount() + 1
    )
    return df

seasonal = compute_pos_ranks(seasonal, "fp_ppr")
seasonal = compute_pos_ranks(seasonal, "fp_half")
6️⃣ Export Final Dataset
final_cols = [
    "player_name",
    "season",
    "position",
    "games",
    "fp_ppr",
    "fp_half",
    "rank_fp_ppr",
    "rank_fp_half"
]

seasonal[final_cols].to_csv(
    "fantasy_season_ranks.csv",
    index=False
)
